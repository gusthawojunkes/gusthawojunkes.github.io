@Service
class PaymentService(
  private val repo: PaymentRepository,
  private val kafka: KafkaTemplate
) {
  @Transactional
  suspend fun process(
    payment: Payment
  ): Result<String> =
    withContext(Dispatchers.IO) {
      repo.save(payment)
        .also { kafka.send(
          "payments",
          it.toEvent()
        )}
        .let { Result.success(it.id) }
    }
